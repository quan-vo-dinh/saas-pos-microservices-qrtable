// libs/frontend/utils/src/lib/__tests__/api-client.spec.ts
import { apiClient, ApiError } from '../api-client';

// Mock global fetch
const mockFetch = jest.fn();
global.fetch = mockFetch;

beforeEach(() => {
  mockFetch.mockReset();
});

describe('apiClient', () => {
  it('should call fetch with correct URL combining baseUrl and path', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => ({ id: '1', name: 'Test' }),
    });

    await apiClient('/categories', { baseUrl: 'http://localhost:3300/api/v1' });

    expect(mockFetch).toHaveBeenCalledWith(
      'http://localhost:3300/api/v1/categories',
      expect.objectContaining({
        headers: expect.objectContaining({ 'Content-Type': 'application/json' }),
      }),
    );
  });

  it('should set Content-Type to application/json by default', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => ({}),
    });

    await apiClient('/test', { baseUrl: '' });

    expect(mockFetch).toHaveBeenCalledWith(
      '/test',
      expect.objectContaining({
        headers: expect.objectContaining({ 'Content-Type': 'application/json' }),
      }),
    );
  });

  it('should merge custom headers with defaults', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => ({}),
    });

    await apiClient('/test', {
      baseUrl: '',
      headers: { 'x-tenant-id': 'tenant_a', Authorization: 'Bearer token123' },
    });

    const calledHeaders = mockFetch.mock.calls[0][1].headers;
    expect(calledHeaders).toEqual(
      expect.objectContaining({
        'Content-Type': 'application/json',
        'x-tenant-id': 'tenant_a',
        Authorization: 'Bearer token123',
      }),
    );
  });

  it('should unwrap BFF response wrapper { data, statusCode, message }', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => ({
        data: [{ id: '1', name: 'Category A' }],
        statusCode: 200,
        message: 'OK',
      }),
    });

    const result = await apiClient<Array<{ id: string; name: string }>>('/categories', { baseUrl: '' });

    expect(result).toEqual([{ id: '1', name: 'Category A' }]);
  });

  it('should return raw payload if not BFF wrapper format', async () => {
    const rawData = { id: '1', name: 'Test' };
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => rawData,
    });

    const result = await apiClient('/test', { baseUrl: '' });

    expect(result).toEqual(rawData);
  });

  it('should throw ApiError on non-ok response', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: false,
      status: 401,
      text: async () => '{"message":"Unauthorized"}',
    });

    await expect(apiClient('/test', { baseUrl: '' })).rejects.toThrow(ApiError);
    await expect(apiClient('/test', { baseUrl: '' })).rejects.toMatchObject({
      status: 401,
    });
  });

  it('should forward HTTP method and body', async () => {
    mockFetch.mockResolvedValueOnce({
      ok: true,
      json: async () => ({ data: { id: '1' }, statusCode: 201, message: 'Created' }),
    });

    await apiClient('/categories', {
      baseUrl: 'http://localhost:3300/api/v1',
      method: 'POST',
      body: JSON.stringify({ name: 'New Category' }),
    });

    expect(mockFetch).toHaveBeenCalledWith(
      'http://localhost:3300/api/v1/categories',
      expect.objectContaining({
        method: 'POST',
        body: JSON.stringify({ name: 'New Category' }),
      }),
    );
  });
});
```

// libs/frontend/utils/src/lib/__tests__/upload-client.spec.ts
import { uploadFile } from '../upload-client';

// Mock XMLHttpRequest
class MockXHR {
  open = jest.fn();
  send = jest.fn();
  setRequestHeader = jest.fn();
  upload = { addEventListener: jest.fn() };
  addEventListener = jest.fn();
  status = 200;
  responseText = '';
}

let mockXhr: MockXHR;

beforeEach(() => {
  mockXhr = new MockXHR();
  jest.spyOn(global, 'XMLHttpRequest').mockImplementation(() => mockXhr as unknown as XMLHttpRequest);
});

afterEach(() => {
  jest.restoreAllMocks();
});

describe('uploadFile', () => {
  const createMockFile = (name = 'test.png', type = 'image/png'): File => new File(['fake-content'], name, { type });

  it('should append file with field name "image" in FormData', () => {
    const file = createMockFile();

    // Start the upload (don't await — we need to inspect the send call)
    uploadFile({ url: 'http://localhost:3300/api/v1/admin/menu-items/123/image', file });

    expect(mockXhr.send).toHaveBeenCalledTimes(1);
    const sentFormData = mockXhr.send.mock.calls[0][0] as FormData;
    expect(sentFormData).toBeInstanceOf(FormData);
    expect(sentFormData.get('image')).toBeTruthy();
    expect(sentFormData.get('file')).toBeNull();
  });

  it('should open POST request to the correct URL', () => {
    const file = createMockFile();
    const url = 'http://localhost:3300/api/v1/admin/menu-items/abc/image';

    uploadFile({ url, file });

    expect(mockXhr.open).toHaveBeenCalledWith('POST', url);
  });

  it('should set custom headers (Authorization, x-tenant-id)', () => {
    const file = createMockFile();
    const headers = {
      Authorization: 'Bearer test-token',
      'x-tenant-id': 'tenant_a',
    };

    uploadFile({ url: 'http://localhost:3300/test', file, headers });

    expect(mockXhr.setRequestHeader).toHaveBeenCalledWith('Authorization', 'Bearer test-token');
    expect(mockXhr.setRequestHeader).toHaveBeenCalledWith('x-tenant-id', 'tenant_a');
  });

  it('should register upload progress listener when onProgress is provided', () => {
    const file = createMockFile();
    const onProgress = jest.fn();

    uploadFile({ url: 'http://localhost:3300/test', file, onProgress });

    expect(mockXhr.upload.addEventListener).toHaveBeenCalledWith('progress', expect.any(Function));
  });

  it('should resolve with parsed response data on success', async () => {
    const file = createMockFile();
    const responseData = { data: { imageUrl: 'https://res.cloudinary.com/test/image.png' } };

    const promise = uploadFile({ url: 'http://localhost:3300/test', file });

    // Simulate successful response
    const loadHandler = mockXhr.addEventListener.mock.calls.find(
      (call: [string, () => void]) => call[0] === 'load',
    )?.[1];
    mockXhr.status = 200;
    mockXhr.responseText = JSON.stringify(responseData);
    loadHandler?.();

    const result = await promise;
    expect(result).toEqual({ imageUrl: 'https://res.cloudinary.com/test/image.png' });
  });

  it('should reject on non-2xx status', async () => {
    const file = createMockFile();

    const promise = uploadFile({ url: 'http://localhost:3300/test', file });

    const loadHandler = mockXhr.addEventListener.mock.calls.find(
      (call: [string, () => void]) => call[0] === 'load',
    )?.[1];
    mockXhr.status = 500;
    mockXhr.responseText = 'Internal Server Error';
    loadHandler?.();

    await expect(promise).rejects.toThrow('Upload failed with status 500');
  });

  it('should reject on network error', async () => {
    const file = createMockFile();

    const promise = uploadFile({ url: 'http://localhost:3300/test', file });

    const errorHandler = mockXhr.addEventListener.mock.calls.find(
      (call: [string, () => void]) => call[0] === 'error',
    )?.[1];
    errorHandler?.();

    await expect(promise).rejects.toThrow('Upload network error');
  });
});
```

// apps/management-app/src/lib/api/__tests__/authenticated-client.spec.ts
import { useAuthStore } from '@/lib/auth/auth-store';

// Mock apiClient from @einvoice/frontend-utils
const mockApiClient = jest.fn();
jest.mock('@einvoice/frontend-utils', () => ({
  apiClient: (...args: unknown[]) => mockApiClient(...args),
}));

// Mock getBffBaseUrl
jest.mock('@/lib/auth/bff-server', () => ({
  getBffBaseUrl: () => 'http://localhost:3300/api/v1',
}));

// Import after mocks
import { authApiClient } from '../authenticated-client';

beforeEach(() => {
  mockApiClient.mockReset();
  mockApiClient.mockResolvedValue({});
  useAuthStore.getState().reset();
});

describe('authApiClient', () => {
  it('should inject Authorization Bearer header from Zustand store', async () => {
    useAuthStore.getState().setAccessToken('test-jwt-token');
    useAuthStore.getState().setProfile({
      userId: '1',
      email: 'test@test.com',
      firstName: 'Test',
      lastName: 'User',
      tenantId: 'tenant_a',
      roles: ['OWNER'],
    });

    await authApiClient('/admin/categories');

    expect(mockApiClient).toHaveBeenCalledWith(
      '/admin/categories',
      expect.objectContaining({
        baseUrl: 'http://localhost:3300/api/v1',
        headers: expect.objectContaining({
          Authorization: 'Bearer test-jwt-token',
          'x-tenant-id': 'tenant_a',
        }),
      }),
    );
  });

  it('should NOT include Authorization header when no accessToken', async () => {
    // Store is reset — no token
    await authApiClient('/admin/categories');

    const calledHeaders = mockApiClient.mock.calls[0][1].headers;
    expect(calledHeaders['Authorization']).toBeUndefined();
  });

  it('should NOT include x-tenant-id header when no profile', async () => {
    await authApiClient('/admin/categories');

    const calledHeaders = mockApiClient.mock.calls[0][1].headers;
    expect(calledHeaders['x-tenant-id']).toBeUndefined();
  });

  it('should forward HTTP method and body', async () => {
    useAuthStore.getState().setAccessToken('token');

    await authApiClient('/admin/categories', {
      method: 'POST',
      body: JSON.stringify({ name: 'New' }),
    });

    expect(mockApiClient).toHaveBeenCalledWith(
      '/admin/categories',
      expect.objectContaining({
        method: 'POST',
        body: JSON.stringify({ name: 'New' }),
      }),
    );
  });

  it('should merge caller-provided headers with auth headers', async () => {
    useAuthStore.getState().setAccessToken('token');
    useAuthStore.getState().setProfile({
      userId: '1',
      email: 'a@b.com',
      firstName: 'A',
      lastName: 'B',
      tenantId: 'tenant_a',
      roles: ['OWNER'],
    });

    await authApiClient('/test', {
      headers: { 'X-Custom': 'value' },
    });

    const calledHeaders = mockApiClient.mock.calls[0][1].headers;
    expect(calledHeaders).toEqual(
      expect.objectContaining({
        Authorization: 'Bearer token',
        'x-tenant-id': 'tenant_a',
        'X-Custom': 'value',
      }),
    );
  });
});

        status: 'available',

    name: 'Drinks',

    items: [],

    expect(result).toEqual([
      { id: 'cat-1', name: 'Appetizers', sortOrder: 0, itemCount: 2 },
      { id: 'cat-2', name: 'Main Course', sortOrder: 1, itemCount: 1 },
      { id: 'cat-3', name: 'Drinks', sortOrder: 2, itemCount: 0 },
    ]);

  it('should return empty array for empty menu', () => {

  });

    const result = extractItems(mockMenu);

    expect(result).toHaveLength(3);
    expect(result.map((i) => i.id)).toEqual(['item-1', 'item-2', 'item-3']);
  });

  it('should return all items when categoryId is null', () => {
    const result = extractItems(mockMenu, null);

    expect(result).toHaveLength(3);
  });

  it('should filter items by categoryId', () => {
    const result = extractItems(mockMenu, 'cat-1');

    expect(result).toHaveLength(2);
    expect(result[0].name).toBe('Spring Rolls');
    expect(result[1].name).toBe('Salad');
  });

  it('should return empty array for unknown categoryId', () => {
    expect(extractItems(mockMenu, 'unknown')).toEqual([]);
